-- Ogni account mantiene la possibilita di agire come Owner.
-- INSERT IGNORE rende il backfill ripetibile senza riattivare capability sospese o disabilitate.
INSERT IGNORE INTO `user_capabilities` (
  `user_id`, `capability`, `status`, `onboarding_version`, `current_step`,
  `completed_at`, `activated_at`, `suspended_at`, `disabled_at`, `created_at`, `updated_at`
)
SELECT
  `id`, 'owner', 'active', 1, 'complete',
  UNIX_TIMESTAMP(), UNIX_TIMESTAMP(), 0, 0, UNIX_TIMESTAMP(), UNIX_TIMESTAMP()
FROM `users`;

FROM `ownerpetgallery`
GROUP BY `pet_id`

LEFT JOIN `booking` AS booking_row ON booking_row.`id` = event.`booking_id`
SET event.`subject_capability` = CASE

UPDATE booking AS b
LEFT JOIN (
  SELECT r.booking_id, MIN(r.created_at) AS first_proposal
  FROM booking_reschedule AS r
  JOIN booking AS original ON original.id = r.booking_id AND original.assistant_user_id = r.proposed_by_user_id
  WHERE r.previous_status = 'pending' AND r.created_at >= original.created_at
  GROUP BY r.booking_id
) AS p ON p.booking_id = b.id
SET b.first_response_at = NULLIF(LEAST(
  COALESCE(p.first_proposal, 4294967295),
  IF(b.accepted_at > 0 AND b.accepted_at >= b.created_at, b.accepted_at, 4294967295),
  IF(b.status = 'declined' AND b.cancelled_by_user_id = b.assistant_user_id AND b.cancelled_at >= b.created_at AND b.cancelled_at > 0, b.cancelled_at, 4294967295)
), 4294967295)
WHERE b.confirmation_mode = 'manual' AND b.first_response_at IS NULL;
